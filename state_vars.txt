      }
      const data = await response.json();
      if (data.text) {

          setRawText(data.text);
        } else {
          setSecondText(data.text);

        }
      } else {
        throw new Error('未讀取到文字內容');
      }

      const errorMsg = err instanceof Error ? err.message : '未知錯誤';
